# Local extraction / repair models. Kept out of requirements.txt so the base install
# stays light — the server boots and every non-ML endpoint works without any of this
# (see app/ml.py, which loads each model lazily and degrades to classical fallbacks).
#
# Install (CUDA 12.x; drop the index-url for a CPU-only box):
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu124
#   pip install --no-deps sam2 simple-lama-inpainting
#   pip install -r requirements-ml.txt
#
# NOTE the --no-deps on sam2: its dependency set pulls `opencv-python`, which collides
# with the `opencv-python-headless` in requirements.txt over the same cv2 module (and on
# Windows the overwrite fails outright if any process has cv2.pyd open). Headless already
# provides everything both packages need here, so sam2's own deps are listed out below
# instead.

torch>=2.4
torchvision>=0.19

# SAM2 — box-prompted segmentation for pixel-perfect element masks.
sam2>=1.1
hydra-core>=1.3
iopath>=0.1.10
tqdm>=4.66

# HQ-SAM (vit_tiny) — the actual box-prompted segmenter as of this line (see
# app/ml.py:get_hqsam_predictor and processing/extract.py:sam2_mask). SAM2's own decoder
# rounds off thin/hollow shapes — a crossed-blade icon's notch and tips came back as a
# blob no matter the prompt (box jitter, iterative mask_input refinement, and box-only vs
# point-augmented all converged to the same wrong shape). HQ-SAM's learnable high-quality
# token fixes exactly this failure mode without hurting anything else measured across the
# ClashUp lobby's other 14 regions. `timm` is HQ-SAM's own dependency for its TinyViT
# backbone. Named "sam2_mask"/"sam2" throughout extract.py for backward compatibility with
# stored region.method values and the auto/classical/sam2 API — see that module.
segment-anything-hq>=0.3
timm>=1.0

# LaMa — inpainting, used to lift icons/text off a frame so it can be reused empty.
simple-lama-inpainting>=0.1.2

# Real-ESRGAN runs through ONNX rather than the basicsr stack, which pins old torch
# and drags in a large transitive tree for one 65MB model.
onnxruntime-gpu>=1.18 ; platform_system != "Darwin"
onnxruntime>=1.18 ; platform_system == "Darwin"
